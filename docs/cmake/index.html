yelu-cmake

Typed surface language that compiles to cmake — with concrete syntax

What is yelu-cmake?

yelu-cmake is the cmake target pack for yelu — and the first test of the compositional language thesis. It provides a typed surface language with concrete syntax that compiles to standard cmake via type erasure. The output is verified for structural equivalence against reference cmake implementations.

cmake is the specimen because it is maximally hostile to automated reasoning: three decades of accumulated layers with no cleanup, implicit namespace collisions, errors that propagate silently as empty strings, and workarounds fossilized as permanent policies. If the thesis holds here, it likely holds for simpler configuration targets.

Code comparison — Tutorial Step 1

cmake

# CMakeLists.txt
cmake_minimum_required(VERSION 3.20)
project(Tutorial)

set(CMAKE_CXX_STANDARD 11)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

configure_file(TutorialConfig.h.in TutorialConfig.h)

add_executable(Tutorial tutorial.cxx)

target_include_directories(Tutorial PUBLIC
  "${PROJECT_BINARY_DIR}"
)

yelu-cmake

# step1.yl
( cmake_minimum_required "3.20";
  project "Tutorial" ~version:"1.0" ~languages:[CXX];

  set 'CMAKE_CXX_STANDARD' "11";
  set 'CMAKE_CXX_STANDARD_REQUIRED' ON;

  configure_file "TutorialConfig.h.in" "TutorialConfig.h";

  let tut : target = Target "Tutorial" in
  add_exe tut { tutorial.cxx };
  include_dirs tut { "${PROJECT_BINARY_DIR}" }
)

Code comparison — Tutorial Step 2

cmake

# CMakeLists.txt
cmake_minimum_required(VERSION 3.20)
project(Tutorial)

add_library(MathFunctions mysqrt.cxx)

add_executable(Tutorial tutorial.cxx)

target_link_libraries(Tutorial PUBLIC MathFunctions)
target_include_directories(Tutorial PUBLIC
  "${PROJECT_BINARY_DIR}"
  "${PROJECT_SOURCE_DIR}/MathFunctions"
)

yelu-cmake

# step2.yl
( cmake_minimum_required "3.20";
  project "Tutorial" ~version:"1.0" ~languages:[CXX];

  let math : target = Target "MathFunctions" in
  add_lib math { mysqrt.cxx };

  let tut : target = Target "Tutorial" in
  add_exe tut { tutorial.cxx };
  link_lib tut { math };
  include_dirs tut {
    "${PROJECT_BINARY_DIR}";
    "${PROJECT_SOURCE_DIR}/MathFunctions"
  }
)

More syntax examples

Conditionals

cmake

if(TARGET Foo)
  message(STATUS "found")
else()
  message(STATUS "not found")
endif()

yelu-cmake

( if target Target Foo then
    ( message 'found' )
  else
    ( message 'not found' )
)

Foreach loops

cmake

foreach(item a b c)
  message("item: ${item}")
endforeach()

yelu-cmake

( foreach item in ["a" "b" "c"]
    ( message "item: ${item}" )
)

Generator expressions

cmake

target_compile_options(Tutorial PRIVATE
  $<$<CONFIG:Debug>:-Wall>
)

install(TARGETS Tutorial
  RUNTIME DESTINATION bin
)

yelu-cmake

( compile_opts Target Tutorial ~private:[$<IF:$<CONFIG:Debug>,-Wall>];

  install_targets [Target Tutorial] "bin"
)

Functions and macros

cmake

function(my_func arg1 arg2)
  message("${arg1}: ${arg2}")
endfunction()

macro(my_macro x y)
  set(${x} ${y})
endmacro()

yelu-cmake

( fun my_func(arg1, arg2)
    ( message '${arg1}: ${arg2}' );

  macro my_macro(x, y)
    ( set '${x}' '${y}' )
)
